from dataclasses import dataclass
from typing import List, Optional

@dataclass
class ResponseHeader:
    content-type: str    #컨텐츠타입
    tr_id: str    #거래ID
    tr_cont: Optional[str] = None    #연속 거래 여부
    gt_uid: Optional[str] = None    #Global UID

@dataclass
class ResponseBody:
    rt_cd: str    #성공 실패 여부
    msg_cd: str    #응답코드
    msg1: str    #응답메세지
    output: List[ResponseBodyoutput] = field(default_factory=list)    #응답상세

@dataclass
class ResponseBodyoutput:
    stac_yymm: str    #결산 년월
    cras: str    #유동자산
    fxas: str    #고정자산
    total_aset: str    #자산총계
    flow_lblt: str    #유동부채
    fix_lblt: str    #고정부채
    total_lblt: str    #부채총계
    cpfn: str    #자본금
    cfp_surp: str    #자본 잉여금
    prfi_surp: str    #이익 잉여금
    total_cptl: str    #자본총계
